\TopicDivider{01}{Hardware Memory Hierarchy \& Cache Invariants}{Latency gaps, spatial locality, and the mechanical sympathy of modern superscalar CPUs.}

\begin{frame}{Hardware Memory Hierarchy}
  \begin{LearningGoals}{Computing Systems Invariants}
    \begin{itemize}\setlength{\itemsep}{.35em}
      \item Quantify CPU cycles lost to DRAM latency versus on-chip SRAM caches (L1/L2/L3).
      \item Identify cache line bouncing and false sharing across multicore SMP architectures.
      \item Contrast Array-of-Structures (AoS) with Structure-of-Arrays (SoA) for SIMD vector throughput.
    \end{itemize}
  \end{LearningGoals}
  \vspace{.15cm}
  \TakeawayBanner{Modern CPUs execute instructions in picoseconds; execution throughput is governed by cache locality.}
\end{frame}

\begin{frame}{Data Layout Architecture: AoS vs SoA}
  \CompareGrid{Array-of-Structures (AoS)}{%
    \texttt{struct Particle \{ float x, y, z; \};}\par
    \texttt{Particle particles[10000];}\par\vspace{.3em}
    \begin{itemize}\setlength{\itemsep}{.15em}
      \item Loading \texttt{x} pulls unneeded \texttt{y, z} into cache.
      \item Pollutes 64-byte cache lines.
      \item Stalls SIMD vector registers on gather.
    \end{itemize}
    \PillBadge[BayesCrimson]{CACHE TRASHING}
  }{Structure-of-Arrays (SoA)}{%
    \texttt{struct ParticleFleet \{}\par
    \texttt{  float x[10000], y[10000], z[10000];}\par
    \texttt{\};}\par\vspace{.3em}
    \begin{itemize}\setlength{\itemsep}{.15em}
      \item Consecutive \texttt{x} floats pack contiguously.
      \item 100\% cache line bandwidth utilization.
      \item Enables 8-wide AVX-512 single-burst vector ops.
    \end{itemize}
    \PillBadge[BayesEmerald]{SIMD OPTIMAL}
  }
\end{frame}
