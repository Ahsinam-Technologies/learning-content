\begin{frame}{Attention Mechanics: The Scaled Dot-Product}
  \MathDerivationStep{1}{Scaled Dot-Product Formula}{\text{Attention}(Q, K, V) = \text{softmax}\left( \frac{Q K^T}{\sqrt{d_k}} \right) V}{%
    For \(q_i, k_j \sim \mathcal{N}(0, 1)\), their dot product has variance \(\text{Var}(q_i \cdot k_j) = d_k\). Dividing by \(\sqrt{d_k}\) preserves unit variance, preventing gradient vanishing in softmax.
  }
  \vspace{.15cm}
  \begin{ConceptCard}[SOFTMAX STABILITY]{Numerical Preservation}
    Large magnitude logits drive softmax derivatives to zero. Scaling by \(\sqrt{d_k}\) maintains healthy gradient flow throughout backpropagation.
  \end{ConceptCard}
\end{frame}
