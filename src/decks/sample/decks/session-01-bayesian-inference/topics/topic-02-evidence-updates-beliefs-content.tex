\TopicDivider{02}{Evidence updates beliefs}{Make the assumptions, data transformation and decision boundary inspectable.}

\begin{frame}[fragile]{A small update should still be reproducible}
  \begin{HighlightedCodeBlock}[numbers=left]{Python}{4,5,6}
prior_alpha, prior_beta = 2, 2
successes, failures = 8, 3

posterior_alpha = prior_alpha + successes
posterior_beta = prior_beta + failures
posterior_mean = posterior_alpha / (posterior_alpha + posterior_beta)
  \end{HighlightedCodeBlock}
  \Citation{The code is short because the model assumption is strong: independent Bernoulli observations and a Beta prior.}
\end{frame}
