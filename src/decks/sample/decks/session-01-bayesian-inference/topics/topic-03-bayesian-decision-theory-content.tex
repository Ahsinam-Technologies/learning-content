\TopicDivider{03}{Bayesian Decision Theory \& Loss Functions}{Translating posterior distributions into optimal operational enterprise policies.}

\begin{frame}{The Principle of Maximum Expected Utility}
  \begin{ConceptCard}[OPTIMAL ACTION]{Loss-Weighted Expectation}
    A posterior distribution \(p(\theta \mid y)\) quantifies belief, but enterprise decisions require taking an action \(a \in \mathcal{A}\). Bayesian Decision Theory pairs probability with an explicit loss function \(L(\theta, a)\):
    \begin{equation*}
      a^* = \arg\min_{a \in \mathcal{A}} \, \mathbb{E}_{\theta \sim p(\theta \mid y)} \big[ L(\theta, a) \big] = \arg\min_{a \in \mathcal{A}} \int_{\Theta} L(\theta, a) \, p(\theta \mid y) \, d\theta
    \end{equation*}
  \end{ConceptCard}
  \vspace{.15cm}
  \TakeawayBanner{Beliefs without loss functions cannot dictate optimal actions; policy requires cost-weighted expectation.}
\end{frame}

\begin{frame}{Canonical Loss Functions \& Optimal Estimators}
  \TwoPane{%
    \begin{DeepDiveCard}[SQUARE LOSS]{Mean Estimation}
      Under \(L(\theta, a) = (\theta - a)^2\), the optimal action is the posterior mean:
      \begin{equation*}
        a^* = \mathbb{E}[\theta \mid y]
      \end{equation*}
      Penalizes large errors quadratically.
    \end{DeepDiveCard}
  }{%
    \begin{DeepDiveCard}[ABSOLUTE LOSS]{Median Estimation}
      Under \(L(\theta, a) = |\theta - a|\), the optimal action is the posterior median:
      \begin{equation*}
        a^* = \text{Median}(\theta \mid y)
      \end{equation*}
      Robust against extreme tail outliers.
    \end{DeepDiveCard}
  }
\end{frame}

\begin{frame}{Optimal Enterprise Decision Thresholds}
  \begin{CautionCard}{Symmetric Loss Fallacy}
    Standard machine learning models optimize symmetric zero-one loss or MSE. In mission-critical production systems, asymmetric failure penalties require shifting the operational classification threshold \(\tau^*\) to the cost-neutral boundary:
    \begin{equation*}
      \tau^* = \frac{C_{\text{FP}}}{C_{\text{FP}} + C_{\text{FN}}} = \frac{5}{5 + 24000} \approx 0.000208
    \end{equation*}
  \end{CautionCard}
  \vspace{.15cm}
  \begin{ConceptCard}[POLICY INVARIANT]{Threshold Governance}
    Triggering fraud investigations at 0.021\% posterior probability minimizes expected enterprise loss, even when 99.9\% of alerts are false alarms.
  \end{ConceptCard}
\end{frame}
