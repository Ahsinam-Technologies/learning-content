\begin{frame}{Concurrency: Mutex Contention vs Lock-Free CAS}
  \CompareGrid{OS Mutex Locks (Pthreads / std::mutex)}{%
    \begin{itemize}\setlength{\itemsep}{.25em}
      \item \textbf{Mechanism}: Acquires exclusive lock; contended threads yield CPU to the OS scheduler.
      \item \textbf{Failure Mode}: Context-switch penalties (1--3\,\(\mu\)s), priority inversion, convoying.
      \item \textbf{Use Case}: Long-duration critical sections with I/O.
    \end{itemize}
    \PillBadge[BayesAmber]{PTHREAD BLOCKING}
  }{Lock-Free Atomics (CAS Primitives)}{%
    \begin{itemize}\setlength{\itemsep}{.25em}
      \item \textbf{Mechanism}: Atomic hardware instructions (\texttt{CMPXCHG16B}) retry without OS kernel context switches.
      \item \textbf{Property}: System-wide progress guaranteed (wait-free / lock-free bounds).
      \item \textbf{Use Case}: Ultra-low-latency queues, ring buffers, atomic counters.
    \end{itemize}
    \PillBadge[BayesEmerald]{HARDWARE NATIVE}
  }
\end{frame}

\begin{frame}{Raft Safety Invariants}
  \AlgorithmCard{Raft Safety Invariants}{O(1) Append, Quorum Verify}{
    \textbf{1. Election Safety:} At most one leader elected per term. \par\vspace{.3em}
    \textbf{2. Leader Append-Only:} Leaders never overwrite logs. \par\vspace{.3em}
    \textbf{3. Log Matching:} If two logs contain identical index/term, all preceding entries match identically.
  }
  \vspace{.15cm}
  \begin{InsightCard}{Linearizable Consistency}
    By requiring quorums to overlap on at least one common node containing the highest term, split-brain partition anomalies are mathematically prohibited.
  \end{InsightCard}
\end{frame}
