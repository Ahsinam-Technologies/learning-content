% --- S2.2 DefinitionCard ---
\begin{frame}{S2.2 — DefinitionCard}
  \begin{DefinitionCard}{Posterior Distribution}
    The conditional probability distribution of the parameter \(\theta\) given the
    observed data \(y\), obtained by combining the prior \(p(\theta)\) and the
    likelihood \(p(y \mid \theta)\) via Bayes' rule and normalising.
  \end{DefinitionCard}
  \BayesCaption{Verify: blue left accent bar, THEORY pill badge, "Definition — " prefix in title.}
\end{frame}

% --- S2.3 InsightCard ---
\begin{frame}{S2.3 — InsightCard}
  \begin{InsightCard}{Posterior is a Distribution, Not a Point}
    The posterior encodes the full residual uncertainty about the parameter after
    observing the data. Reducing it to a single estimate discards calibration
    information that matters for downstream decisions.
  \end{InsightCard}
  \BayesCaption{Verify: teal left accent bar, CORE pill badge.}
\end{frame}

% --- S2.4 CautionCard ---
\begin{frame}{S2.4 — CautionCard}
  \begin{CautionCard}{Do Not Confuse Likelihood with Prior}
    A likelihood \(p(y \mid \theta)\) is not a distribution over \(\theta\).
    Treating it as one before multiplying by a prior and normalising is a
    common source of incorrect inference.
  \end{CautionCard}
  \BayesCaption{Verify: amber left accent bar, WATCHOUT pill badge.}
\end{frame}

% --- S2.5 WorkedExample ---
\begin{frame}{S2.5 — WorkedExample}
  \begin{WorkedExample}{Beta-Binomial Conjugate Update}
    Given \(\theta \sim \text{Beta}(\alpha_0, \beta_0)\) and \(n\) Bernoulli trials
    with \(k\) successes, the posterior is:
    \[
      \theta \mid k \sim \text{Beta}(\alpha_0 + k,\; \beta_0 + n - k).
    \]
    With \(\alpha_0 = \beta_0 = 1\) (uniform prior) and \(k = 7, n = 10\):
    posterior is \(\text{Beta}(8, 4)\).
  \end{WorkedExample}
  \BayesCaption{Verify: plum left accent bar, APPLIED pill badge.}
\end{frame}

% --- S2.6 KeyTakeaway ---
\begin{frame}{S2.6 — KeyTakeaway}
  \begin{KeyTakeaway}{Normalisation Makes the Posterior a Probability}
    Dividing by the marginal likelihood \(p(y)\) ensures the posterior integrates to
    one. Omitting this step leaves the numerator as an unnormalised measure, not
    a distribution.
  \end{KeyTakeaway}
  \BayesCaption{Verify: teal left accent bar, SUMMARY pill badge, pale background.}
\end{frame}

% --- S2.7 ProofSketch ---
\begin{frame}{S2.7 — ProofSketch}
  \begin{ProofSketch}{Existence of Conjugate Posteriors}
    If the prior and likelihood belong to an exponential family with conjugate
    structure, the posterior is in the same family with updated natural parameters.
    The sufficient statistics accumulate additively across i.i.d. observations,
    making sequential updates computationally identical to a single batch update.
  \end{ProofSketch}
  \BayesCaption{Verify: dark (BayesDark) left accent bar, RIGOR pill badge.}
\end{frame}

% --- S2.8 ExerciseCard ---
\begin{frame}{S2.8 — ExerciseCard}
  \begin{ExerciseCard}{Derive the Posterior Mean for Beta-Binomial}
    \begin{enumerate}
      \item Write out the unnormalised posterior \(p(\theta \mid k) \propto p(k \mid \theta)\,p(\theta)\).
      \item Identify the kernel of the Beta distribution.
      \item Compute the posterior mean \(\mathbb{E}[\theta \mid k]\) in terms of \(\alpha, \beta, n, k\).
    \end{enumerate}
  \end{ExerciseCard}
  \BayesCaption{Verify: teal left accent bar, LAB pill badge, pale background.}
\end{frame}

  \MathDerivationStep{1}{Bayes' Rule}{%
    p(\theta \mid y) = \frac{p(y \mid \theta)\,p(\theta)}{p(y)}
  }{%
    The marginal \(p(y)\) normalises the product of likelihood and prior to produce
    a proper probability distribution over \(\theta\).
  }

% --- S3.3 Two MathDerivationSteps stacked (vertical fit test) ---
% Expected: both steps fit without overflowing into the footer.
\begin{frame}{S3.3 — Two MathDerivationSteps (vertical fit test)}
  \MathDerivationStep{1}{Continuous SDE}{%
    dS_t = \mu S_t\,dt + \sigma S_t\,dW_t
  }{%
    Asset prices follow a continuous diffusion where \(W_t\) is a standard Wiener
    process satisfying \(\mathbb{E}[dW_t] = 0\), \(\mathrm{Var}(dW_t) = dt\).
  }
  \BayesGap
  \MathDerivationStep{2}{It\^{o}'s Chain Rule}{%
    df = \!\left(\frac{\partial f}{\partial t} + \mu S\frac{\partial f}{\partial S}
    + \tfrac{1}{2}\sigma^2 S^2\frac{\partial^2 f}{\partial S^2}\right)\!dt
    + \sigma S\frac{\partial f}{\partial S}\,dW_t
  }{%
    The quadratic variation \((dW_t)^2 = dt\) introduces the non-zero second-order
    drift correction absent from classical calculus.
  }
  \BayesCaption{Verify: both steps fully visible; no footer overlap.}
\end{frame}

% --- S3.4 ProofSketch with inline math ---
\begin{frame}{S3.4 — ProofSketch with inline math}
  \begin{ProofSketch}{Ito Isometry}
    For an adapted square-integrable process \(f(t)\):
    \[
      \mathbb{E}\!\left[\left(\int_0^T\! f(t)\,dW_t\right)^{\!2}\right]
      = \mathbb{E}\!\left[\int_0^T\! f(t)^2\,dt\right].
    \]
    The isometry converts a stochastic integral into a Lebesgue integral,
    enabling \(L^2\) analysis of Ito processes.
  \end{ProofSketch}
  \BayesCaption{Verify: display math inside ProofSketch is readable and not clipped.}
\end{frame}

def log_posterior(theta, y, alpha, beta):
    log_prior = (alpha - 1)*np.log(theta) + (beta - 1)*np.log(1 - theta)
    log_lik   = np.sum(y * np.log(theta) + (1 - y) * np.log(1 - theta))

\begin{frame}{S11.3 — Single Large Component}
  \begin{WorkedExample}{Full KV Cache Memory Calculation}
    \textbf{Given:} LLaMA-3 70B, 8k context, BF16 precision, 80 layers, \(d_\text{model}=8192\).\par\vspace{.4em}
    \textbf{Per layer:}
    \[
      \text{Cache} = 2 \times 2 \times d_\text{model} \times N_\text{ctx}
      = 2 \times 2 \times 8192 \times 8192 \approx 268\,\text{MB per layer}
    \]
    \textbf{Total:}
    \[
      80 \times 268\,\text{MB} \approx 21.4\,\text{GB per request}
    \]
    \textbf{Reduction with GQA (8 KV heads vs 64 Q heads):}
    \[
      21.4\,\text{GB} \times (8/64) \approx 1.2\,\text{GB per request}
    \]
    Grouped-Query Attention reduces KV cache by the Q:KV head ratio,
    enabling higher batch sizes at the same GPU memory budget.
  \end{WorkedExample}
\end{frame}
